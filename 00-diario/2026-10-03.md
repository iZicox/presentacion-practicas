- para poder insertar el codigo del inventario por consultar la api la mejor forma es crear un shortcode e insertarlo en la plantilla de pagina de producto

```php
// Registrar el shortcode [stock_api]
add_shortcode('stock_api', 'mostrar_stock_dummyjson_shortcode');

function mostrar_stock_dummyjson_shortcode()
{
    // 1. Obtener el objeto global de producto de WooCommerce
    global $product;

    // Verificar si estamos en una página de producto válida de WooCommerce
    if (!is_a($product, 'WC_Product')) {
        return '';
    }

    $product_id = $product->get_id();

    // 2. Gestionar la caché de WordPress (Transient)
    $transient_key = 'stock_api_shortcode_' . $product_id;
    $stock = get_transient($transient_key);

    // Si no está guardado en caché, consultar la API
    if (false === $stock) {

        // Endpoint de prueba (DummyJSON ID 1)
        // Para hacerlo dinámico en el futuro: 'https://dummyjson.com/products/' . $product_id
        $generarId = random_int(1, 30);
        $api_url = 'https://dummyjson.com/products/' . $generarId;

        // Petición HTTP remota desde el servidor
        $response = wp_remote_get($api_url, array('timeout' => 5));

        // Si la petición a la API falla o da error de conexión
        if (is_wp_error($response)) {
            return '<div class="stock-api-box error" style="color: #e74c3c; font-weight: bold; margin: 10px 0;">Stock: No disponible (Error de conexión)</div>';
        }

        $body = wp_remote_retrieve_body($response);
        $data = json_decode($body, true);

        // Extraer la clave "stock" del JSON devuelto
        if (isset($data['stock'])) {
            $stock = intval($data['stock']);

            // Guardar en la caché de WordPress durante 15 minutos (900 segundos)
            set_transient($transient_key, $stock, 15 * MINUTE_IN_SECONDS);
        } else {
            return '<div class="stock-api-box error" style="color: #e74c3c; font-weight: bold; margin: 10px 0;">Producto no encontrado en inventario</div>';
        }
    }

    // 3. Devolver el marcado HTML según la cantidad de stock
    if ($stock > 0) {
        return '<div class="stock-api-box disponible" style="color: #27ae60; font-weight: bold; margin: 10px 0; font-size: 1.05em;">' .
            '✔ Stock disponible en fábrica: ' . esc_html($stock) . ' unidades' .
            '</div>';
    } else {
        return '<div class="stock-api-box agotado" style="color: #e74c3c; font-weight: bold; margin: 10px 0; font-size: 1.05em;">' .
            '✖ Sin stock disponible / Bajo pedido' .
            '</div>';
    }
}

```

//functions.php
<?php
/**
 * Hello Elementor Child Theme functions and definitions
 */

// Encolar los estilos del tema padre y del tema hijo
add_action( 'wp_enqueue_scripts', 'hello_elementor_child_enqueue_styles', 20 );

function hello_elementor_child_enqueue_styles() {
    // Carga el estilo del tema padre
    wp_enqueue_style( 'hello-elementor', get_template_directory_uri() . '/style.css' );
    
    // Carga el estilo personalizado del tema hijo
    wp_enqueue_style( 'hello-elementor-child', get_stylesheet_uri(), array( 'hello-elementor' ), '1.0.0' );
}
